'use client';

import { useEffect, useState } from 'react';
import { createClient } from '@/lib/supabase/client';

interface ActionData {
  code: string;
  variation_pct: number;
  capitalisation: number;
  cours_jour: number;
}

export default function HeatmapMarket() {
  const [actions, setActions] = useState<ActionData[]>([]);
  const [loading, setLoading] = useState(true);
  const supabase = createClient();

  useEffect(() => {
    const fetchData = async () => {
      const { data } = await supabase
        .from('brvm_actions_daily')
        .select('code, variation_pct, capitalisation, cours_jour')
        .order('capitalisation', { ascending: false });
      setActions(data || []);
      setLoading(false);
    };

    fetchData();
    const interval = setInterval(fetchData, 15 * 60 * 1000); // 15 min
    return () => clearInterval(interval);
  }, [supabase]);

  if (loading) return <div className="text-muted">Chargement heatmap...</div>;

  const maxCap = Math.max(...actions.map((a) => a.capitalisation || 0));
  const minCap = Math.min(...actions.map((a) => a.capitalisation || 0));

  return (
    <div className="grid grid-cols-8 gap-2 p-4">
      {actions.map((a) => {
        const var_pct = a.variation_pct || 0;
        const hue = var_pct > 0 ? 120 : var_pct < 0 ? 0 : 0;
        const sat = Math.abs(var_pct) > 10 ? 100 : Math.abs(var_pct) * 10;
        const light = 50 - (Math.abs(var_pct) / 10) * 10;
        const bgColor = `hsl(${hue}, ${sat}%, ${light}%)`;
        const size = (a.capitalisation / maxCap) * 100 + 50; // 50px to 150px
        return (
          <div
            key={a.code}
            className="rounded border border-border flex items-center justify-center cursor-pointer transition hover:scale-105 hover:shadow-lg"
            style={{ 
              backgroundColor: bgColor, 
              width: `${size}px`, 
              height: `${size}px` 
            }}
            title={`${a.code}: ${var_pct >= 0 ? '+' : ''}${var_pct.toFixed(2)}% · ${a.cours_jour?.toLocaleString('fr-FR')} FCFA`}
          >
            <div className="text-center">
              <p className="text-xs font-bold text-white">{a.code}</p>
              <p className="text-[10px] text-white">{var_pct >= 0 ? '+' : ''}{var_pct.toFixed(2)}%</p>
            </div>
          </div>
        );
      })}
    </div>
  );
}

// frontend/lib/screener/filters.ts
import type { ScreenerPreset } from './presets';

export interface ActionRow {
  code: string;
  cours_jour: number;
  variation_pct: number;
  rsi: number;
  volume_moyen: number;
  score_signal: number;
  secteur: string;
  rendement_dividende: number | null;
}

export function applyFilters(
  actions: ActionRow[],
  filters: ScreenerPreset['filters']
): ActionRow[] {
  return actions.filter((a) => {
    if (filters.rsiMin != null && a.rsi < filters.rsiMin) return false;
    if (filters.rsiMax != null && a.rsi > filters.rsiMax) return false;
    if (filters.volumeGtAvg && a.volume_moyen === 0) return false;
    if (filters.scoreMin != null && a.score_signal < filters.scoreMin) return false;
    if (filters.secteurs?.length && !filters.secteurs.includes(a.secteur)) return false;
    if (filters.dividendMin != null && (a.rendement_dividende ?? 0) < filters.dividendMin) return false;
    return true;
  });
}

// frontend/components/ScreenerFilters.tsx
'use client';

import { useState } from 'react';
import { PRESETS } from '@/lib/screener/presets';

interface FiltersState {
  rsiMin: number;
  rsiMax: number;
  scoreMin: number;
  secteur: string;
  dividendMin: number;
}

export default function ScreenerFilters({
  isPremium,
  onFilterChange,
}: {
  isPremium: boolean;
  onFilterChange: (filters: FiltersState) => void;
}) {
  const [filters, setFilters] = useState<FiltersState>({
    rsiMin: 0,
    rsiMax: 100,
    scoreMin: 0,
    secteur: '',
    dividendMin: 0,
  });

  const handleChange = (key: keyof FiltersState, value: number | string) => {
    const newFilters = { ...filters, [key]: value };
    setFilters(newFilters);
    onFilterChange(newFilters);
  };

  return (
    <div className="bg-surface border border-border rounded-xl p-4 space-y-4">
      <div className="space-y-2">
        <label className="text-xs text-muted uppercase">RSI</label>
        <div className="flex gap-2">

function useGetProfile() {
  const [profile, setProfile] = useState<any>(null);
  const supabase = createClient();

      supabase.from('profiles').select('is_premium').eq('id', user?.id).maybeSingle().then(({ data }) => setProfile(data));
    });
  }, [supabase]);
  return { data: profile };
}

// frontend/lib/dividend/calendar.ts
export interface DividendEvent {
  code: string;
  date_detachement: string; // YYYY-MM-DD
  taux: number;
  rendement_estime: number;
  date_paiement: string;
}

export function getDaysUntil(dateStr: string): number {
  const target = new Date(dateStr);
  const today = new Date();
  const diff = target.getTime() - today.getTime();
  return Math.ceil(diff / (1000 * 60 * 60 * 24));
}

export function getMonthDays(year: number, month: number): number {
  return new Date(year, month + 1, 0).getDate();
}

export function groupByMonth(dividends: DividendEvent[]): Map<string, DividendEvent[]> {
  const map = new Map<string, DividendEvent[]>();
  dividends.forEach((d) => {
    const month = d.date_detachement.substring(0, 7); // YYYY-MM
    if (!map.has(month)) map.set(month, []);
    map.get(month)!.push(d);
  });
  return map;
}

'use client';

import { useState, useEffect } from 'react';
import { createClient } from '@/lib/supabase/client';
import { getDaysUntil, groupByMonth } from '@/lib/dividend/calendar';
import type { DividendEvent } from '@/lib/dividend/calendar';

export default function DividendCalendar() {
  const [dividends, setDividends] = useState<DividendEvent[]>([]);
  const [hoveredDiv, setHoveredDiv] = useState<DividendEvent | null>(null);
  const [month, setMonth] = useState(new Date());

        .from('brvm_dividends')
        .select('code, date_detachement, taux, rendement_estime, date_paiement');

    };
    fetch();
  }, [supabase]);

  const year = month.getFullYear();
  const monthIdx = month.getMonth();
  const firstDay = new Date(year, monthIdx, 1).getDay();

      </div>

      <div className="grid grid-cols-7 gap-2">
        {cells.map((day, i) => {
          const dateStr = day ? `${year}-${String(monthIdx + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}` : null;
          const divForDay = dateStr ? monthDivs.find((d) => d.date_detachement === dateStr) : null;
          const daysUntil = divForDay ? getDaysUntil(dateStr!) : 0;

          return (
            <div
              key={i}
              className={`aspect-square rounded border flex items-center justify-center text-sm transition ${
                divForDay
                  ? 'border-cyan/40 bg-cyan/10 text-cyan cursor-pointer hover:bg-cyan/20'

import { supabase } from '../persistence/supabase.js';
import pino from 'pino';

const log = pino({ name: 'paperTradingService' });

export async function createPosition(
  userId: string,
  accountId: string,
  code: string,
  entryPrice: number,
  entryDate: string,
  size: number,
  signalId?: string
) {
  const { data, error } = await supabase
    .from('paper_trading_positions')
    .insert({
      user_id: userId,
      account_id: accountId,
      code,
      entry_price: entryPrice,
      entry_date: entryDate,
      entry_signal_id: signalId,
      size,
      status: 'open',
    });
  if (error) throw new Error(`createPosition: ${error.message}`);
  return data?.[0];
}

export async function closePosition(
  positionId: string,
  exitPrice: number,
  exitDate: string,
  signalId?: string
) {
  const { data: pos } = await supabase
    .from('paper_trading_positions')
    .select('entry_price, size, entry_date')
    .eq('id', positionId)
    .maybeSingle();
  if (!pos) throw new Error('Position not found');

  const pnl = (exitPrice - pos.entry_price) * pos.size;
  const pnl_pct = ((exitPrice - pos.entry_price) / pos.entry_price) * 100;
  const days_held = Math.floor(
    (new Date(exitDate).getTime() - new Date(pos.entry_date).getTime()) / (1000 * 60 * 60 * 24)
  );

  const { error } = await supabase
    .from('paper_trading_positions')
    .update({
      exit_price: exitPrice,
      exit_date: exitDate,
      exit_signal_id: signalId,
      status: 'closed',
      pnl,
      pnl_pct,
      days_held,
    })
    .eq('id', positionId);
  if (error) throw new Error(`closePosition: ${error.message}`);

  await updateAccountMetrics(pos.user_id);
  return { pnl, pnl_pct, days_held };
}

export async function updateAccountMetrics(userId: string) {
  const { data: positions } = await supabase
    .from('paper_trading_positions')
    .select('pnl, status')
    .eq('user_id', userId);

  const totalPnl = positions?.reduce((sum, p) => sum + (p.pnl ?? 0), 0) ?? 0;
  const { data: account } = await supabase
    .from('paper_trading_accounts')
    .select('capital_initial')
    .eq('user_id', userId)
    .maybeSingle();

  const capitalCurrent = (account?.capital_initial ?? 0) + totalPnl;
  const pnlPct = account?.capital_initial ? (totalPnl / account.capital_initial) * 100 : 0;

  await supabase
    .from('paper_trading_accounts')
    .update({
      pnl_total: totalPnl,
      pnl_pct: pnlPct,
      capital_current: capitalCurrent,
    })
    .eq('user_id', userId);

  log.info({ userId, totalPnl, pnlPct }, 'Account metrics updated');
}

export async function autoClosePositions() {
  const today = new Date().toISOString().split('T')[0];
  const { data: oldPositions } = await supabase
    .from('paper_trading_positions')
    .select('id, user_id, code')
    .eq('status', 'open')
    .lt('created_at', new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString());

  for (const pos of oldPositions ?? []) {
    const { data: daily } = await supabase
      .from('brvm_actions_daily')
      .select('cours_jour')
      .eq('code', pos.code)
      .eq('date_marche', today)
      .maybeSingle();

    if (daily) {
      await closePosition(pos.id, daily.cours_jour, today);
      log.info({ positionId: pos.id }, 'Position auto-closed after 30 days');
    }
  }
}

import { createClient } from '@/lib/supabase/server';
import { NextResponse } from 'next/server';

export async function GET() {
  const supabase = createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });

  const { data: positions } = await supabase
    .from('paper_trading_positions')
    .select('pnl_pct, status, days_held')
    .eq('user_id', user.id);

  const closed = positions?.filter((p) => p.status === 'closed') ?? [];
  const wins = closed.filter((p) => (p.pnl_pct ?? 0) > 0).length;
  const winRate = closed.length > 0 ? ((wins / closed.length) * 100).toFixed(1) : '0';
  const avgWin = closed.filter((p) => (p.pnl_pct ?? 0) > 0).reduce((sum, p) => sum + (p.pnl_pct ?? 0), 0) / (wins || 1);
  const avgLoss = closed.filter((p) => (p.pnl_pct ?? 0) <= 0).reduce((sum, p) => sum + (p.pnl_pct ?? 0), 0) / (closed.length - wins || 1);
  const avgDaysHeld = closed.reduce((sum, p) => sum + (p.days_held ?? 0), 0) / (closed.length || 1);

  return NextResponse.json({
    totalTrades: closed.length,
    winRate,
    avgWin: avgWin.toFixed(2),
    avgLoss: avgLoss.toFixed(2),
    avgDaysHeld: Math.round(avgDaysHeld),
  });
}

    const doc = new PDFDocument({ size: 'A4', margin: 40 });
    const chunks: Buffer[] = [];

    doc.on('data', (chunk) => chunks.push(chunk));
    doc.on('end', () => resolve(Buffer.concat(chunks)));
    doc.on('error', reject);

    // Page 1: Résumé
    doc.fontSize(24).text('BRVM Analyst Pro', { align: 'center' });
    doc.fontSize(14).text(`Rapport de synthèse — ${content.month}`, { align: 'center' });
    doc.moveDown();

    doc.fontSize(12).text('Résumé du mois', { underline: true });
    doc.fontSize(10).text(`P&L portefeuille: ${content.pnlTotal.toLocaleString('fr-FR')} FCFA (${content.pnlPct.toFixed(2)}%)`);
    doc.text(`Email: ${content.userEmail}`);
    doc.moveDown();

    // Page 2: Top signaux
    doc.addPage().fontSize(12).text('Top 3 signaux gagnants', { underline: true });
    content.topSignals.forEach((s, i) => {
      doc.fontSize(10).text(`${i + 1}. ${s.code} — P&L ${s.pnlPct.toFixed(2)}%`);
    });
    doc.moveDown();
    doc.fontSize(9).text(content.narration.signals, { align: 'left' });

    // Page 3: Analyse fondamentale
    doc.addPage().fontSize(12).text('Analyse fondamentale', { underline: true });
    content.fundamentals.forEach((f) => {
      doc.fontSize(10).text(`${f.code} — Graham: ${f.graham.toFixed(0)}, DCF: ${f.dcf.toFixed(0)}, PER: ${f.per.toFixed(1)}`);
    });
    doc.moveDown();
    doc.fontSize(9).text(content.narration.fundamentals);

    // Page 4: Événements
    doc.addPage().fontSize(12).text('Événements du mois', { underline: true });
    content.events.forEach((e) => {
      doc.fontSize(10).text(`${e.code} (${e.type}): ${e.description}`);
    });
    doc.moveDown();
    doc.fontSize(9).text(content.narration.events);

    doc.end();
  });
}

// scraper/src/tasks/generateMonthlyReports.ts
import { supabase } from '../persistence/supabase.js';
import { generatePDF } from '../services/pdfGenerator.js';
import { narrateReport } from '../services/reportNarrator.js';
import pino from 'pino';

const log = pino({ name: 'generateMonthlyReports' });

export async function generateMonthlyReports() {
  log.info('Starting monthly report generation');

  const month = new Date().toISOString().slice(0, 7); // YYYY-MM
  const { data: premiumUsers } = await supabase
    .from('profiles')
    .select('id, email, is_premium')
    .eq('is_premium', true);

  for (const user of premiumUsers ?? []) {
    try {
      // 1. Fetch paper trading data
      const { data: account } = await supabase
        .from('paper_trading_accounts')
        .select('pnl_total, pnl_pct')
        .eq('user_id', user.id)
        .maybeSingle();

      const { data: positions } = await supabase
        .from('paper_trading_positions')
        .select('code, entry_price, exit_price, pnl_pct')
        .eq('user_id', user.id)
        .eq('status', 'closed')
        .order('pnl_pct', { ascending: false })
        .limit(3);

      const topSignals = positions?.map((p) => ({
        code: p.code,
        entry: p.entry_price,
        exit: p.exit_price,
        pnl: 0,
        pnlPct: p.pnl_pct ?? 0,
      })) ?? [];

      // 2. Fetch events
      const { data: events } = await supabase
        .from('brvm_news')
        .select('*')
        .gte('date_publication', `${month}-01`)
        .lte('date_publication', `${month}-31`)
        .limit(5);

      // 3. Call IA for narration
      const narration = await narrateReport({
        topSignals,
        fundamentals: [],
        events: events?.map((e) => ({ code: e.instrument_code, type: 'news' })) ?? [],
      }, process.env.DEEPSEEK_API_KEY!);

      // 4. Generate PDF
      const pdfBuffer = await generatePDF({
        userEmail: user.email,
        month,
        pnlTotal: account?.pnl_total ?? 0,
        pnlPct: account?.pnl_pct ?? 0,
        topSignals,
        fundamentals: [],
        events: events?.map((e) => ({ code: e.instrument_code, type: 'news', description: e.titre })) ?? [],
        narration,
      });

      // 5. Upload & save
      const reportUrl = `https://brvm.s3.amazonaws.com/reports/${user.id}/${month}.pdf`;
      // TODO: Upload to S3 or Vercel Blob

      await supabase.from('monthly_reports').insert({
        user_id: user.id,
        month,
        report_url: reportUrl,
        report_json: { topSignals, narration },
        sent_at: new Date().toISOString(),
      });

      log.info({ userId: user.id, month }, 'Report generated');
    } catch (err) {
      log.error({ userId: user.id, err }, 'Failed to generate report');
    }
  }
}

import { createClient } from '@/lib/supabase/server';
import { NextRequest, NextResponse } from 'next/server';

export async function GET(
  req: NextRequest,
  { params }: { params: { month: string } }
) {
  const supabase = createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });

  const { data: report } = await supabase
    .from('monthly_reports')
    .select('*')
    .eq('user_id', user.id)
    .eq('month', params.month)
    .maybeSingle();

  if (!report) return NextResponse.json({ error: 'Report not found' }, { status: 404 });
  return NextResponse.json(report);
}
